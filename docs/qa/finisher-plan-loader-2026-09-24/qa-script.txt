import { chromium } from '/Users/troy/.claude/worktrees/finisher-plan-loader/node_modules/playwright-core/index.mjs';
import { encode } from '/Users/troy/.claude/worktrees/finisher-plan-loader/node_modules/next-auth/jwt.js';
import { mkdir, writeFile } from 'node:fs/promises';
import assert from 'node:assert/strict';
const out='/Users/troy/.claude/worktrees/finisher-plan-loader/docs/qa/finisher-plan-loader-2026-09-24';
await mkdir(out,{recursive:true});
const browser=await chromium.launch({headless:true});
const results=[];
try {
for(const [label,width,height] of [['desktop',1440,900],['mobile',390,844]]) {
 const suffix=label==='desktop'?'':'-mobile';
 const context=await browser.newContext({viewport:{width,height},recordVideo:{dir:out,size:{width,height}}});
 const token=await encode({secret:'finisher-local-ui-qa-only-not-production',salt:'authjs.session-token',token:{sub:'local-loader-qa',name:'Local QA'},maxAge:3600});
 await context.addCookies([{name:'authjs.session-token',value:token,url:'http://127.0.0.1:3013',httpOnly:true,sameSite:'Lax'}]);
 const page=await context.newPage();
 const errors=[]; const failed=[];
 page.on('pageerror', e=>errors.push(e.message));
 page.on('console',m=>{if(m.type()==='error') errors.push(m.text());});
 page.on('requestfailed',r=>failed.push({url:r.url(),error:r.failure()?.errorText}));
 await page.goto('http://127.0.0.1:3013/onboarding');
 await page.getByRole('heading',{name:"When's your race?"}).waitFor();
 const next=page.getByRole('button',{name:'Continue',exact:true});
 assert.equal(await next.isEnabled(),false);
 const date=page.locator('input[type=date]');
 await date.fill(new Date(Date.now()+7*86400000).toISOString().slice(0,10));
 assert.equal(await next.isEnabled(),false);
 await page.getByText('Race is less than 4 weeks away.',{exact:false}).waitFor();
 await page.screenshot({path:out+'/R5'+suffix+'-validation.png'});
 await date.fill(new Date(Date.now()+84*86400000).toISOString().slice(0,10));
 await page.getByPlaceholder('e.g. London, Manchester, Sydney').fill('Local QA');
 await next.click();
 await page.getByRole('heading',{name:'Your fitness baseline'}).waitFor();
 assert.equal(await next.isEnabled(),true);
 await page.getByPlaceholder('MM:SS (e.g. 28:00)').fill('28:00');
 await next.click();
 await page.getByRole('heading',{name:'About you'}).waitFor();
 assert.equal(await next.isEnabled(),true);
 await page.getByPlaceholder('e.g. 32').fill('32');
 await page.getByRole('button',{name:'Prefer not to say',exact:true}).click();
 await next.click();
 await page.getByRole('heading',{name:'Training availability'}).waitFor();
 await page.getByRole('button',{name:'3 days Minimum effective'}).click();
 await page.getByRole('button',{name:'45 min',exact:true}).click();
 await next.click();
 await page.getByRole('heading',{name:'Where are you right now?'}).waitFor();
 await page.locator('button').filter({has:page.locator('svg.lucide-chevron-left')}).click();
 await page.getByRole('heading',{name:'Training availability'}).waitFor();
 assert.match(await page.getByRole('button',{name:'45 min',exact:true}).getAttribute('class'),/border-primary/);
 await next.click();
 await page.getByRole('button',{name:/Decent fitness/}).click();
 const submit=page.getByRole('button',{name:'Build My Plan',exact:true});
 assert.equal(await submit.isEnabled(),true);
 await page.screenshot({path:out+'/R5'+suffix+'.png'});
 const started=Date.now(); await submit.click();
 const status=page.getByRole('status'); await status.waitFor({timeout:1500});
 assert.equal(await page.getByRole('button',{name:'Build My Plan',exact:true}).count(),0);
 const style=await status.evaluate(el=>({bg:getComputedStyle(el.parentElement).backgroundColor,rect:el.parentElement.getBoundingClientRect().toJSON(),color:getComputedStyle(el.querySelector('svg')).color,animation:getComputedStyle(el.querySelector('svg')).animationName,transform:getComputedStyle(el.querySelector('svg')).transform}));
 assert.equal(style.bg,'rgb(10, 10, 10)'); assert.equal(style.color,'rgb(34, 197, 94)'); assert.equal(style.animation,'spin'); assert.equal(style.rect.width,width); assert.equal(style.rect.height,height);
 await page.screenshot({path:out+'/R1'+suffix+'.png'});
 const transform2=await status.locator('svg').evaluate(el=>getComputedStyle(el).transform);
 assert.notEqual(style.transform,transform2);
 await page.screenshot({path:out+'/R2'+suffix+'.png'});
 await page.waitForURL('**/dashboard');
 await page.getByRole('status').filter({hasText:'Building your plan...'}).waitFor({state:'detached'});
 await page.screenshot({path:out+'/R3'+suffix+'.png'});
 results.push({label,style,transform2,url:page.url(),elapsed:Date.now()-started,errors,failed});
 assert.deepEqual(errors,[]);
 await context.close();
}
await writeFile(out+'/runtime.json',JSON.stringify(results,null,2));
process.stdout.write(JSON.stringify(results,null,2)+'\nPASS desktop and mobile onboarding, validation, full-screen loader, animation, dashboard redirect\n');
} finally {await browser.close();}
