"use client"
import { useState } from "react"
import { StepRace } from "@/components/app/onboarding/step-race"
import { weeksUntilRace } from "@/components/ui/date-picker"
import DashboardPage from "@/app/(app)/dashboard/page"
import { usePlanStore } from "@/stores/plan-store"
import ProfilePage from "@/app/(app)/profile/page"
import { useAthleteStore } from "@/stores/athlete-store"
import type { AthleteProfile } from "@/lib/types"
const fixture: AthleteProfile = { raceDate: "2027-03-01", location: "Manila", category: "open", fiveKTime: "28:00", tenKTime: "", hyroxTime: "", age: 30, gender: "", weight: 70, weightUnit: "kg", daysPerWeek: 4, sessionLength: 60, fitnessLevel: "beginner_mid" }
export default function CalendarHarness() {
 const [data, setData] = useState<Partial<AthleteProfile>>({ category: "open" })
 const [dashboardMode, setDashboardMode] = useState(false)
 const [profileMode, setProfileMode] = useState(false)
 const weeks = data.raceDate ? weeksUntilRace(data.raceDate) : 0
 return <main className="mx-auto max-w-xl p-4">
  <p className="mb-4 text-sm text-muted-foreground">Local component fixture — no authenticated session</p>
  <button className="min-h-11 border px-4 mb-4" onClick={() => { useAthleteStore.getState().setProfile(fixture); setProfileMode(!profileMode) }}>Toggle profile fixture</button>
  <button className="min-h-11 border px-4 mb-4" onClick={() => { useAthleteStore.getState().clearProfile(); usePlanStore.getState().clearPlan(); setDashboardMode(true) }}>Show dashboard fixture</button>
  {dashboardMode ? <DashboardPage /> : profileMode ? <ProfilePage /> : <>
    <StepRace data={data} onChange={value => setData(previous => ({ ...previous, ...value }))} />
    {data.raceDate && weeks < 4 && <p role="alert">Race is less than 4 weeks away. We need at least 4 weeks for a meaningful plan.</p>}
    <button disabled={!data.raceDate || weeks < 4} className="mt-6 min-h-11 w-full bg-primary text-primary-foreground disabled:opacity-40">Continue</button>
    <output>{data.raceDate}</output>
  </>}
 </main>
}
